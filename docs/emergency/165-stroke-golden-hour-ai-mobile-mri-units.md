### 4.1. Spatial Feature Transform (SFT)
The generator incorporates a Spatial Feature Transform (SFT) layer. Instead of applying uniform global reconstruction, the network is conditioned on a prior probability map of stroke lesions (derived from diffusion-weighted imaging / DWI sequences). This ensures that the boundaries of ischemic penumbras and hemorrhagic pools are preserved with sub-millimeter accuracy.

Where:
- $\mathcal{L}_{\text{grad}}$ is the **Gradient Difference Loss (GDL)**, which prevents the synthesized images from being overly smooth and preserves sharp structural edges.

import torch
import torch.nn as nn
import torch.nn.functional as F
from typing import Tuple, Dict

class SpatialFeatureTransform(nn.Module):
    """
    Spatial Feature Transform (SFT) layer to inject stroke-aware prior features
    into the low-field MRI reconstruction pipeline.
    """
    def __init__(self, in_channels: int, prior_channels: int = 64):
        super().__init__()
        self.conv_gamma = nn.Sequential(
            nn.Conv3d(prior_channels, in_channels, kernel_size=3, padding=1),
            nn.LeakyReLU(0.2, inplace=True),
            nn.Conv3d(in_channels, in_channels, kernel_size=3, padding=1)
        )
        self.conv_beta = nn.Sequential(
            nn.Conv3d(prior_channels, in_channels, kernel_size=3, padding=1),
            nn.LeakyReLU(0.2, inplace=True),
            nn.Conv3d(in_channels, in_channels, kernel_size=3, padding=1)
        )

    def forward(self, x: torch.Tensor, prior: torch.Tensor) -> torch.Tensor:
        # x: [B, C, D, H, W]
        # prior: [B, C_prior, D, H, W]
        gamma = self.conv_gamma(prior)
        beta = self.conv_beta(prior)
        return x * (1.0 + gamma) + beta

class ResidualBlock3D(nn.Module):
    """
    3D Residual Block with Batch Normalization and ReLU activation.
    """
    def __init__(self, channels: int):
        super().__init__()
        self.conv1 = nn.Conv3d(channels, channels, kernel_size=3, padding=1, bias=False)
        self.bn1 = nn.BatchNorm3d(channels)
        self.relu = nn.ReLU(inplace=True)
        self.conv2 = nn.Conv3d(channels, channels, kernel_size=3, padding=1, bias=False)
        self.bn2 = nn.BatchNorm3d(channels)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        residual = x
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += residual
        return self.relu(out)

class StrokeAwareGenerator3D(nn.Module):
    """
    3D Stroke-Aware CycleGAN Generator for reconstructing 0.064T ULF MRI to 3T equivalent.
    Integrates a spatial feature transform (SFT) to preserve critical ischemic/hemorrhagic boundaries.
    """
    def __init__(self, in_channels: int = 1, out_channels: int = 1, num_features: int = 64, num_blocks: int = 6):
        super().__init__()
        
        # Initial downsampling / feature extraction
        self.initial = nn.Sequential(
            nn.Conv3d(in_channels, num_features, kernel_size=7, stride=1, padding=3, bias=False),
            nn.BatchNorm3d(num_features),
            nn.ReLU(inplace=True)
        )
        
        # Downsampling layers
        self.down1 = nn.Sequential(
            nn.Conv3d(num_features, num_features * 2, kernel_size=3, stride=2, padding=1, bias=False),
            nn.BatchNorm3d(num_features * 2),
            nn.ReLU(inplace=True)
        )
        self.down2 = nn.Sequential(
            nn.Conv3d(num_features * 2, num_features * 4, kernel_size=3, stride=2, padding=1, bias=False),
            nn.BatchNorm3d(num_features * 4),
            nn.ReLU(inplace=True)
        )

        # Prior network to extract stroke-specific spatial priors (e.g., from DWI/FLAIR metadata)
        self.prior_net = nn.Sequential(
            nn.Conv3d(in_channels, 32, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv3d(32, 64, kernel_size=3, stride=4, padding=1), # Downsample to match bottleneck resolution
            nn.ReLU(inplace=True)
        )

        # Bottleneck with Residual Blocks and Spatial Feature Transform
        self.sft = SpatialFeatureTransform(num_features * 4, prior_channels=64)
        self.res_blocks = nn.ModuleList([ResidualBlock3D(num_features * 4) for _ in range(num_blocks)])

        # Upsampling layers
        self.up1 = nn.Sequential(
            nn.ConvTranspose3d(num_features * 4, num_features * 2, kernel_size=3, stride=2, padding=1, output_padding=1, bias=False),
            nn.BatchNorm3d(num_features * 2),
            nn.ReLU(inplace=True)
        )
        self.up2 = nn.Sequential(
            nn.ConvTranspose3d(num_features * 2, num_features, kernel_size=3, stride=2, padding=1, output_padding=1, bias=False),
            nn.BatchNorm3d(num_features),
            nn.ReLU(inplace=True)
        )
        
        # Output layer
        self.output = nn.Sequential(
            nn.Conv3d(num_features, out_channels, kernel_size=7, stride=1, padding=3),
            nn.Tanh()
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Input shape: [B, 1, D, H, W]
        feat = self.initial(x)
        feat = self.down1(feat)
        feat = self.down2(feat)
        
        # Extract spatial prior and apply SFT
        prior = self.prior_net(x)
        feat = self.sft(feat, prior)
        
        # Pass through residual blocks
        for block in self.res_blocks:
            feat = block(feat)
            
        # Upsample to original resolution
        feat = self.up1(feat)
        feat = self.up2(feat)
        return self.output(feat)

class ResNet3DClassifier(nn.Module):
    """
    3D ResNet Classifier to differentiate Ischemic vs Hemorrhagic stroke
    and output a confidence score.
    """
    def __init__(self, in_channels: int = 1, num_classes: int = 2):
        super().__init__()
        self.conv = nn.Conv3d(in_channels, 32, kernel_size=3, stride=2, padding=1, bias=False)
        self.bn = nn.BatchNorm3d(32)
        self.relu = nn.ReLU(inplace=True)
        
        self.layer1 = ResidualBlock3D(32)
        self.layer2 = nn.Sequential(
            nn.Conv3d(32, 64, kernel_size=3, stride=2, padding=1, bias=False),
            nn.BatchNorm3d(64),
            nn.ReLU(inplace=True),
            ResidualBlock3D(64)
        )
        
        self.pool = nn.AdaptiveAvgPool3d((1, 1, 1))
        self.fc = nn.Linear(64, num_classes)

    def forward(self, x: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        # Input shape: [B, 1, D, H, W]
        out = self.relu(self.bn(self.conv(x)))
        out = self.layer1(out)
        out = self.layer2(out)
        out = self.pool(out)
        out = torch.flatten(out, 1)
        logits = self.fc(out)
        probs = F.softmax(logits, dim=1)
        
        # Class 0: Hemorrhagic, Class 1: Ischemic
        confidence, stroke_class = torch.max(probs, dim=1)
        return stroke_class, confidence

    Unified Stroke Diagnostic Pipeline combining 3D reconstruction and classification.
    """

        return {
            "reconstructed_scan": high_res_scan,
            "stroke_type_idx": stroke_type_idx,
            "confidence": confidence
        }
