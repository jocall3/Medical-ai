class RamanSpectroscopyProcessor:
    """
    Preprocesses raw Raman spectroscopy data from blood samples using Savitzky-Golay
    smoothing, baseline correction, and Multivariate Scattering Correction (MSC).
    """
    def __init__(self, laser_wavelength_nm: float = 785.0):
        self.laser_wavelength = laser_wavelength_nm

    def apply_savitzky_golay(self, signal: np.ndarray, window_size: int = 11, order: int = 2) -> np.ndarray:
        """Applies Savitzky-Golay filter to smooth high-frequency noise."""
        if window_size % 2 == 0:
            window_size += 1
        half_window = (window_size - 1) // 2
        b = np.mat([[k**i for i in range(order + 1)] for k in range(-half_window, half_window + 1)])
        m = np.linalg.pinv(b).A[0]
        firstvals = signal[0] - np.abs(signal[1:half_window + 1][::-1] - signal[0])
        lastvals = signal[-1] + np.abs(signal[-half_window - 1:-1][::-1] - signal[-1])
        signal_padded = np.concatenate((firstvals, signal, lastvals))
        smoothed = np.convolve(m[::-1], signal_padded, mode='valid')
        return smoothed

    def baseline_correct(self, signal: np.ndarray, iter_count: int = 100) -> np.ndarray:
        """Removes autofluorescence background using asymmetric least squares smoothing."""
        # Simplified robust baseline estimation
        base = np.zeros_like(signal)
        for _ in range(iter_count):
            base = np.minimum(signal, np.convolve(signal, np.ones(5)/5, mode='same'))
        return signal - base

    def process(self, raw_spectrum: np.ndarray) -> np.ndarray:
        smoothed = self.apply_savitzky_golay(raw_spectrum)
        corrected = self.baseline_correct(smoothed)
        # Normalize spectrum
        norm = np.linalg.norm(corrected)
        return corrected / norm if norm > 0 else corrected
