   * Computes the Kolmogorov-Smirnov distance between real-time biometrics and baseline.
   */
  private computeKSDistance(current: number[], baseline: number[]): number {
    const sortedCurrent = [...current].sort((a, b) => a - b);
    const sortedBaseline = [...baseline].sort((a, b) => a - b);
    
    let maxD = 0;
    const totalPoints = Math.max(sortedCurrent.length, sortedBaseline.length);

    for (let i = 0; i < totalPoints; i++) {
      const eCDFCurrent = sortedCurrent.filter(x => x <= sortedCurrent[i]).length / sortedCurrent.length;
      const eCDFBaseline = sortedBaseline.filter(x => x <= sortedCurrent[i]).length / sortedBaseline.length;
      const d = Math.abs(eCDFCurrent - eCDFBaseline);
      if (d > maxD) {
        maxD = d;
      }
    }
    return maxD;
  }

   * Calculates the Shannon Entropy of the ambient audio stream to detect micro-vocalizations of pain.
   */

    const frequencies: { [key: number]: number } = {};
    pitchData.forEach(p => {
      frequencies[p] = (frequencies[p] || 0) + 1;
    });

    let entropy = 0;
    const total = pitchData.length;
    for (const key in frequencies) {
      const p = frequencies[key] / total;
      entropy -= p * Math.log2(p);
    }
    return entropy;
  }
